Python Example
# Make Payment Request import requests def make_payment(): url = 'https://universe-wallet.vercel.app/api/payment' params = { 'key': 'YOUR_KEY', 'number': '9876543210', 'amount': 500, 'comment': 'Monthly Payment' } try: response = requests.get(url, params=params) data = response.json() if data.get('status') == 'success': print(f'Payment Successful! TXN ID: {data.get("transaction_id")}') else: print(f'Payment Failed: {data.get("message")}') except requests.exceptions.RequestException as e: print(f'Request Failed: {e}')
