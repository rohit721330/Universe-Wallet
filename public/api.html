JAVASCRIPT EXAMPLE
// Make Payment Request — Universal Wallet async function makePayment() { const apiUrl = 'https://universe-wallet.vercel.app/api/payment'; const params = new URLSearchParams({ key: 'YOUR_KEY', number: '9876543210', amount: 500, comment: 'Monthly Payment' }); try { const response = await fetch(`${apiUrl}?${params}`); const data = await response.json(); if (data.status === 'success') { console.log('Payment Successful:', data); } else { console.error('Payment Failed:', data.message); } } catch (error) { console.error('API Error:', error); } }
PYTHON EXAMPLE
# Make Payment Request — Universal Wallet import requests def make_payment(): url = 'https://universe-wallet.vercel.app/api/payment' params = { 'key': 'YOUR_KEY', 'number': '9876543210', 'amount': 500, 'comment': 'Monthly Payment' } try: response = requests.get(url, params=params) data = response.json() if data.get('status') == 'success': print(f'Payment Successful! TXN ID: {data.get("transaction_id")}') else: print(f'Payment Failed: {data.get("message")}') except requests.exceptions.RequestException as e: print(f'Request Failed: {e}')
